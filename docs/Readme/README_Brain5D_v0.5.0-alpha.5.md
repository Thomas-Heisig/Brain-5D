### 1. src/homeostasis/heatmap.py – Homeostasis‑Heatmap‑Source


python
"""Heatmap source for homeostasis metrics (rate error, threshold, energy)."""

from dataclasses import dataclass
from typing import Optional

import numpy as np

from src.core.network import NeuralNetwork
from src.core.spatial_index import unpack_coords


@dataclass(frozen=True)
class HomeostasisHeatmapResult:
    """Result of a homeostasis heatmap build."""
    kind: str          # 'rate_error', 'threshold', 'energy_error'
    values: np.ndarray # 2D array (X × Y) projected
    min_val: float
    max_val: float
    shape: tuple[int, int]


class HomeostasisHeatmapSource:
    """Builds heatmaps from homeostasis state of a neural network."""

    def __init__(self, network: NeuralNetwork, dimensions: tuple[int, int, int, int, int]):
        self.network = network
        self.dims = dimensions

    def build_rate_error(self) -> HomeostasisHeatmapResult:
        """Heatmap of rate error (actual - target) per neuron, projected to X-Y."""
        target_rate = self.network.config.get("homeostasis", {}).get("target_rate_hz", 5.0)
        heatmap = np.zeros((self.dims[0], self.dims[1]), dtype=np.float64)
        counts = np.zeros_like(heatmap, dtype=np.float64)

        for nid, neuron in self.network.neurons.items():
            x, y, _, _, _ = unpack_coords(nid)
            # actual rate from homeostasis engine (stored in neuron?)
            # For now, we approximate by recent spike count / tau
            # In real implementation, the HomeostasisEngine should store a running rate per neuron.
            # We'll compute a smoothed rate from spike_counter and current tick.
            # For demo, we use a placeholder: if neuron has a rate attribute, use it.
            rate = getattr(neuron, "_smoothed_rate", 0.0)
            if rate == 0.0:
                # fallback: use spike_counter / tick
                tick = max(1, self.network.current_tick)
                rate = neuron.spike_counter / tick
            error = rate - target_rate
            heatmap[x, y] += error
            counts[x, y] += 1.0

        counts[counts == 0.0] = 1.0
        heatmap /= counts
        return HomeostasisHeatmapResult(
            kind="rate_error",
            values=heatmap,
            min_val=float(np.min(heatmap)),
            max_val=float(np.max(heatmap)),
            shape=heatmap.shape,
        )

    def build_threshold(self) -> HomeostasisHeatmapResult:
        """Heatmap of threshold adaptation per neuron."""
        heatmap = np.zeros((self.dims[0], self.dims[1]), dtype=np.float64)
        counts = np.zeros_like(heatmap, dtype=np.float64)

        for nid, neuron in self.network.neurons.items():
            x, y, _, _, _ = unpack_coords(nid)
            adapt = getattr(neuron, "threshold_adaptation", 0.0)
            heatmap[x, y] += adapt
            counts[x, y] += 1.0

        counts[counts == 0.0] = 1.0
        heatmap /= counts
        return HomeostasisHeatmapResult(
            kind="threshold",
            values=heatmap,
            min_val=float(np.min(heatmap)),
            max_val=float(np.max(heatmap)),
            shape=heatmap.shape,
        )

    def build_energy_error(self) -> HomeostasisHeatmapResult:
        """Heatmap of energy deviation from target."""
        target_energy = self.network.config.get("homeostasis", {}).get("target_energy", 1.0)
        heatmap = np.zeros((self.dims[0], self.dims[1]), dtype=np.float64)
        counts = np.zeros_like(heatmap, dtype=np.float64)

        for nid, neuron in self.network.neurons.items():
            x, y, _, _, _ = unpack_coords(nid)
            error = neuron.energy - target_energy
            heatmap[x, y] += error
            counts[x, y] += 1.0

        counts[counts == 0.0] = 1.0
        heatmap /= counts
        return HomeostasisHeatmapResult(
            kind="energy_error",
            values=heatmap,
            min_val=float(np.min(heatmap)),
            max_val=float(np.max(heatmap)),
            shape=heatmap.shape,
        )

def update(self, tick: int) -> None:
    if not self.enabled:
        return

    tau = self.rate_tau_ticks
    alpha = 1.0 / tau if tau > 0 else 1.0
    target = self.target_rate

    for neuron in self.network.neurons.values():
        # Smoothed rate
        if not hasattr(neuron, "_smoothed_rate"):
            neuron._smoothed_rate = 0.0
        # Spike detection: we need to know if it spiked this tick.
        # We can read the last_spike_tick and compare to current tick.
        spiked = (neuron.last_spike_tick == tick)
        rate = neuron._smoothed_rate + alpha * (1.0 if spiked else 0.0 - neuron._smoothed_rate)
        neuron._smoothed_rate = rate

        # Threshold adaptation
        error = rate - target
        delta = self.threshold_lr * error
        new_adapt = neuron.threshold_adaptation + delta
        # clamp
        new_adapt = max(self.threshold_min, min(self.threshold_max, new_adapt))
        neuron.threshold_adaptation = new_adapt

        # Energy homeostasis
        if self.energy_enabled:
            energy_error = self.target_energy - neuron.energy
            neuron.energy += self.energy_recovery_rate * energy_error
            neuron.energy = max(self.energy_min, min(self.energy_max, neuron.energy))

def _serve_homeostasis_heatmap(
    self,
    server: DashboardServer,
    query: dict[str, list[str]],
) -> None:
    """Serve homeostasis heatmap (rate_error, threshold, energy_error)."""
    from src.homeostasis.heatmap import HomeostasisHeatmapSource
    # Wir benötigen Zugriff auf das Netzwerk – wir speichern es im Server oder State.
    # Alternativ: Der DashboardStateStore enthält das Netzwerk.
    network = server.dashboard_state._network  # (vorausgesetzt, wir speichern es)
    if network is None:
        self._send_json({"error": "No network available"}, HTTPStatus.SERVICE_UNAVAILABLE)
        return
    source = HomeostasisHeatmapSource(network, network.dimensions)
    kind = query.get("kind", ["rate_error"])[0]
    if kind == "rate_error":
        result = source.build_rate_error()
    elif kind == "threshold":
        result = source.build_threshold()
    elif kind == "energy_error":
        result = source.build_energy_error()
    else:
        self._send_json({"error": f"Unknown kind: {kind}"}, HTTPStatus.BAD_REQUEST)
        return
    self._send_json({
        "kind": result.kind,
        "values": result.values.tolist(),
        "min": result.min_val,
        "max": result.max_val,
        "shape": result.shape,
    })

        "network": {"initial_connections_per_neuron": 10, "neighbour_radius": 2.0},
        "topology": {"allow_self_connections": False, "allow_parallel_connections": False},
    }
    rng = random.Random(config["seed"])
    net = NeuralNetwork(config, rng)
    # initialisiere Netzwerk (vereinfacht)
    total_positions = 100000
    idx = rng.sample(range(total_positions), config["initial_neurons"])
    for i in idx:
        coord = linear_to_5d(i, config["dimensions"])
        net.add_neuron(coord)
    net.initialize_random_connections(10, 2.0)

    engine = HomeostasisEngine(net, config)
    engine.attach(net)

    # 100.000 Ticks simuliert
    for tick in range(100000):
        net.step()
        # engine.update wird via Hook aufgerufen

    rates = [getattr(n, "_smoothed_rate", 0.0) for n in net.neurons.values()]

    energies = [n.energy for n in net.neurons.values()]

Persistence contract

Brain-5D currently uses four coordinated persistence layers:

.b5d Snapshot
      +
State Delta Journal
      +
Structural Journal
      +
Runtime Checkpoint
Snapshot

The frozen .b5d V1 format provides:

compact binary records;
deterministic layout;
memory mapping;
random access;
sorted neuron/synapse records;
optical sidecar support.
State delta journal

The state journal provides:

append-only delta records;
CRC protection;
commit markers;
monotonic tick handling;
recovery from uncommitted tails;
crash-safe replay.
Structural journal

Alpha.5 adds a separate append-only structural journal for:

neuron additions;
neuron removals;
synapse additions;
synapse removals;
committed CRC-protected records;
monotonic sequence numbers;
deterministic replay;
persistent inverse undo.

Undo never deletes history. It appends an inverse structural record.

Runtime checkpoint

The runtime checkpoint preserves state that is not fully represented by the frozen
snapshot alone, including:

current tick;
total spikes;
total processed events;
RNG state;
pending currents;
queued events;
input/output cells;
exact floating-point continuation state.
Restore order

The current restore lifecycle is:

1. Load .b5d snapshot
2. Apply committed state-delta journal records
3. Replay committed structural journal records
4. Restore runtime checkpoint
5. Validate consistency
6. Continue simulation

The structural replay occurs before the runtime checkpoint is overlaid, preserving the
expected network topology for deterministic continuation.

Structural approval

Structural changes are deliberately conservative.

Default safety posture:

self_organization:
  enabled: false
  dry_run: true
  auto_approval: false
  allow_neuron_pruning: false

This means:

self-organization does not mutate the network by default;
auto approval is OFF;
neuron pruning is OFF;
structural changes require an explicitly enabled and valid mutation path.

Auto approval is only permitted when all configured confidence, cooldown and safety
limits are satisfied.

Snapshot behavior

A dashboard snapshot request does not write arbitrary state immediately in the HTTP
request thread.

The runtime controller records the request and processes it at a safe runtime boundary.

The intended persistence ordering is:

Structural Journal flush/commit
        ↓
.b5d Snapshot
        ↓
Runtime Checkpoint
        ↓
Dashboard completion/status update

This prevents the operator interface from capturing a topology in the middle of a
controlled structural mutation batch.

Type-safety policy

Brain-5D treats type safety as an architectural requirement.

Current rules include:

no unparameterized dict, list, tuple, deque, etc.;
JSON/YAML data enters the system as untrusted object and is narrowed explicitly;
no dynamic runtime attributes on neuron objects;
no broad Any escape hatches;
no broad type: ignore / pyright: ignore;
subprocess arguments use explicit typed argument lists;
configuration keys must exist in the declared typed contracts;
dashboard code must not reach into private network/engine attributes;
enum handling should be exhaustive;
mutable internals stay behind typed public interfaces.

Shared contracts should live in or reuse:

src/typing_contracts.py
